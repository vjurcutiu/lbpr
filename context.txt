nginx.dev:
worker_processes auto;

events { worker_connections 1024; }

http {
  include       mime.types;
  sendfile      on;
  tcp_nopush    on;
  tcp_nodelay   on;
  keepalive_timeout 65;

  # WebSocket upgrade helper
  map $http_upgrade $connection_upgrade { default upgrade; '' close; }

  upstream fastapi_upstream { server api:8000; keepalive 64; }
  upstream vite_upstream    { server spa:5173; keepalive 64; }

  server {
    listen 80;
    server_name app.localhost;

    # ---------- API → FastAPI ----------
    location /api/ {
      proxy_pass http://fastapi_upstream/;
      proxy_set_header Host              $host;
      proxy_set_header X-Real-IP         $remote_addr;
      proxy_set_header X-Forwarded-For   $proxy_add_x_forwarded_for;
      proxy_set_header X-Forwarded-Proto $scheme;
      proxy_http_version 1.1;
      proxy_redirect off;
      proxy_read_timeout 1h;
      proxy_send_timeout 1h;
      proxy_buffering off;   # keep API responses streaming in dev
      proxy_cache off;
    }

    # ---------- App → Vite dev server (HMR) ----------
    location / {
      proxy_pass http://vite_upstream;
      proxy_http_version 1.1;

      # HMR websocket upgrade
      proxy_set_header Upgrade    $http_upgrade;
      proxy_set_header Connection $connection_upgrade;

      proxy_set_header Host              $host;
      proxy_set_header X-Real-IP         $remote_addr;
      proxy_set_header X-Forwarded-For   $proxy_add_x_forwarded_for;
      proxy_set_header X-Forwarded-Proto $scheme;

      proxy_read_timeout 1h;
      proxy_send_timeout 1h;

      # Critical for fast HMR in dev
      proxy_buffering off;
      proxy_cache off;
      gzip off;
    }
  }
}

import { defineConfig } from "vite";
import react from "@vitejs/plugin-react";
import tailwind from "@tailwindcss/vite";
import path from "path";

const rootDir = process.cwd();
const isDockerDev = process.env.DOCKER_DEV === "1";

export default defineConfig({
  plugins: [react(), tailwind()],

  resolve: { alias: { "@": path.resolve(rootDir, "src") } },

  server: {
    host: true,          // 0.0.0.0 inside the container
    port: 5173,
    strictPort: true,

    // HMR goes through Nginx at app.localhost:80
    hmr: isDockerDev
      ? {
          host: "app.localhost", // browser-visible host
          protocol: "ws",        // "wss" if Nginx terminates TLS
          clientPort: 80,        // key when reverse-proxying HMR
        }
      : true,

    // Prefer NO polling for speed; only enable if your host FS requires it
    // watch: isDockerDev ? { usePolling: true, interval: 150 } : undefined,

    proxy: {}, // keep API on Nginx (/api → FastAPI)
  },

  esbuild: { target: "es2020" },

  // Let Vite keep a stable prebundle cache inside the image/volume
  optimizeDeps: {
    force: false,
  },
});
