# Add these GitHub secrets so deploys can run the API with Doppler:
#
# DOPPLER_TOKEN          -> Service Token scoped to your project/config (read-only)
# DOPPLER_PROJECT        -> e.g. lbpr (Doppler project slug)
# DOPPLER_CONFIG         -> e.g. prod (or staging)
#
# The workflow will write them into server file ./.deploy.env so compose can pass them to the API container.

name: CI — Build, Push, and Deploy (Doppler)

on:
  push:
    branches:
      - main
      - develop
  workflow_dispatch:

env:
  REGISTRY: docker.io

permissions:
  contents: read
  packages: write
  id-token: write

jobs:
  build-and-deploy:
    runs-on: ubuntu-latest

    steps:
      - name: Checkout repository
        uses: actions/checkout@v5

      - name: Set up QEMU
        uses: docker/setup-qemu-action@v3

      - name: Set up Docker Buildx
        uses: docker/setup-buildx-action@v3

      - name: Log in to Docker Hub
        uses: docker/login-action@v3
        with:
          username: ${{ secrets.DOCKERHUB_USERNAME }}
          password: ${{ secrets.DOCKERHUB_TOKEN }}

      - name: Validate Doppler secrets
        run: |
          set -euo pipefail
          if [ -z "${{ secrets.DOPPLER_TOKEN_API }}" ]; then
            echo "DOPPLER_TOKEN_API is not set" >&2
            exit 1
          fi
          if [ -z "${{ secrets.DOPPLER_TOKEN_SPA }}" ]; then
            echo "DOPPLER_TOKEN_SPA is not set" >&2
            exit 1
          fi
          if [ -z "${{ secrets.DOPPLER_PROJECT }}" ]; then
            echo "DOPPLER_PROJECT is not set" >&2
            exit 1
          fi
          if [ -z "${{ secrets.DOPPLER_CONFIG }}" ]; then
            echo "DOPPLER_CONFIG is not set" >&2
            exit 1
          fi

      - name: Compute image tags (sha-*)
        id: tags
        run: |
          set -euo pipefail
          SHORT=$(echo "${GITHUB_SHA}" | cut -c1-7)
          echo "api_tag=sha-${SHORT}" >> "$GITHUB_OUTPUT"
          echo "spa_tag=sha-${SHORT}" >> "$GITHUB_OUTPUT"
          echo "short=${SHORT}" >> "$GITHUB_OUTPUT"

      - name: Build and push API image
        uses: docker/build-push-action@v6
        with:
          context: ./backend
          file: ./backend/Dockerfile
          push: true
          tags: |
            ${{ env.REGISTRY }}/${{ secrets.DOCKERHUB_USERNAME }}/lbpr-api:${{ steps.tags.outputs.api_tag }}

      - name: Install Doppler CLI (SPA)
        uses: dopplerhq/cli-action@v3

      - name: Generate SPA .env from Doppler
        working-directory: static/spa
        env:
          DOPPLER_TOKEN: ${{ secrets.DOPPLER_TOKEN_SPA }}
        run: |
          set -euo pipefail
          doppler secrets download --no-file --format env > .env.production
          echo "SPA .env.production generated with $(wc -l < .env.production) entries"

      - name: Validate SPA env (Firebase)
        working-directory: static/spa
        run: |
          set -euo pipefail
          if [ ! -f .env.production ]; then
            echo "[SPA ENV] ERROR: .env.production not found" >&2
            ls -la
            exit 1
          fi
          required_vars=(
            VITE_FIREBASE_API_KEY
            VITE_FIREBASE_AUTH_DOMAIN
            VITE_FIREBASE_PROJECT_ID
            VITE_FIREBASE_APP_ID
            VITE_FIREBASE_MESSAGING_SENDER_ID
          )
          echo "[SPA ENV] Validating Firebase vars in .env.production"
          for var in "${required_vars[@]}"; do
            line=$(grep "^$var=" .env.production || true)
            value=${line#"$var="}
            if [ -z "$line" ]; then
              echo "[SPA ENV] ERROR: $var is missing in .env.production" >&2
              exit 1
            fi
            if [ -z "$value" ]; then
              echo "[SPA ENV] ERROR: $var is empty in .env.production" >&2
              exit 1
            fi
            echo "[SPA ENV] $var present (length ${#value})"
          done
          echo "[SPA ENV] Firebase env validation OK"

      - name: Build and push SPA image
        uses: docker/build-push-action@v6
        with:
          context: ./static/spa
          file: ./static/spa/Dockerfile
          push: true
          tags: |
            ${{ env.REGISTRY }}/${{ secrets.DOCKERHUB_USERNAME }}/lbpr-spa:${{ steps.tags.outputs.spa_tag }}
          build-args: |
            CI_FAST_BUILD=1

      - name: Prepare deploy bundle
        run: |
          set -euo pipefail

          mkdir -p _deploy/ops/deploy
          mkdir -p _deploy/ops/nginx/snippets
          mkdir -p _deploy/ops/certs
          mkdir -p _deploy/ops/redis
          mkdir -p _deploy/backend
          mkdir -p _deploy/static

          cp docker-compose.yml _deploy/
          if [ -f docker-compose.ssl.yml ]; then cp docker-compose.ssl.yml _deploy/; fi

          # SPA marketing assets, if present
          if [ -d static/marketing ]; then
            cp -R static/marketing _deploy/static/
          fi

          # Nginx config + snippets for production
          # Ensure nginx.conf in deploy bundle is always a regular file
          rm -rf _deploy/ops/nginx/nginx.conf
          cp ops/nginx/nginx.conf _deploy/ops/nginx/nginx.conf
          if [ -f ops/nginx/snippets/cloudflare-real-ip.conf ]; then
            cp ops/nginx/snippets/cloudflare-real-ip.conf _deploy/ops/nginx/snippets/cloudflare-real-ip.conf
          fi

          # SSL certs and ACME webroot
          mkdir -p _deploy/ops/certs/cf-origin

          # Cloudflare Origin cert + key from GitHub secrets (lexbot.pro)
          printf "%s\n" "${{ secrets.CF_ORIGIN_CERT_LEXBOT_PRO }}" > _deploy/ops/certs/cf-origin/lexbot.pro-origin.pem
          printf "%s\n" "${{ secrets.CF_ORIGIN_KEY_LEXBOT_PRO }}" > _deploy/ops/certs/cf-origin/lexbot.pro-origin.key

          # Optional ACME webroot (if you also use certbot for Let's Encrypt)
          if [ -d ops/certbot-www ]; then
            cp -R ops/certbot-www _deploy/ops/
          fi

          # Redis prod config
          if [ -f ops/redis/redis.prod.conf ]; then
            cp ops/redis/redis.prod.conf _deploy/ops/redis/redis.prod.conf
          fi

          # Compose deploy overlays
          cp ops/deploy/docker-compose.deploy.yml _deploy/ops/deploy/docker-compose.deploy.yml
          cp ops/deploy/docker-compose.doppler.yml _deploy/ops/deploy/docker-compose.doppler.yml
          cp ops/deploy/deploy.sh _deploy/ops/deploy/deploy.sh
          chmod +x _deploy/ops/deploy/deploy.sh

          # Root .env (Redis, shared secrets) from GitHub secret
          printf "%s" "${{ secrets.SERVER_ROOT_ENV }}" > _deploy/.env

          # Backend .env from GitHub secret
          printf "%s" "${{ secrets.SERVER_BACKEND_ENV }}" > _deploy/backend/.env

          # Firebase service account (optional, base64-encoded JSON)
          if [ -n "${{ secrets.FIREBASE_SA_JSON_B64 }}" ]; then
            echo "${{ secrets.FIREBASE_SA_JSON_B64 }}" | base64 -d > _deploy/backend/lexbot-pro-firebase-adminsdk-fbsvc-40d434171d.json
          fi

      - name: Start ssh-agent
        uses: webfactory/ssh-agent@v0.9.0
        with:
          ssh-private-key: ${{ secrets.HETZNER_SSH_KEY }}

      - name: Add Hetzner host to known_hosts
        env:
          HETZNER_SSH_PORT: ${{ secrets.HETZNER_SSH_PORT }}
          HETZNER_HOST: ${{ secrets.HETZNER_HOST }}
        run: |
          mkdir -p ~/.ssh
          ssh-keyscan -p "${HETZNER_SSH_PORT:-22}" "${HETZNER_HOST}" >> ~/.ssh/known_hosts

      - name: Fix remote nginx.conf path (ensure not a directory)
        env:
          HETZNER_SSH_PORT: ${{ secrets.HETZNER_SSH_PORT }}
          HETZNER_USER: ${{ secrets.HETZNER_USER }}
          HETZNER_HOST: ${{ secrets.HETZNER_HOST }}
          REMOTE_DIR: ${{ secrets.REMOTE_DIR }}
        run: |
          ssh -p "${HETZNER_SSH_PORT:-22}" "${HETZNER_USER}@${HETZNER_HOST}" "
            set -euo pipefail
            cd \"${REMOTE_DIR:-/opt/lbpr}\"
            if [ -d ops/nginx/nginx.conf ] && [ ! -f ops/nginx/nginx.conf ]; then
              echo '[deploy] WARNING: ops/nginx/nginx.conf is a directory on remote; removing so rsync can lay down a file'
              rm -rf ops/nginx/nginx.conf
            fi
          "

      - name: Rsync deploy bundle to server
        env:
          HETZNER_SSH_PORT: ${{ secrets.HETZNER_SSH_PORT }}
          HETZNER_USER: ${{ secrets.HETZNER_USER }}
          HETZNER_HOST: ${{ secrets.HETZNER_HOST }}
          REMOTE_DIR: ${{ secrets.REMOTE_DIR }}
        run: |
          set -euo pipefail
          RSYNC_RSH="ssh -p ${HETZNER_SSH_PORT:-22}" rsync -avz --delete _deploy/ "${HETZNER_USER}@${HETZNER_HOST}:${REMOTE_DIR:-/opt/lbpr}/"

      - name: Remote deploy
        env:
          HETZNER_SSH_PORT: ${{ secrets.HETZNER_SSH_PORT }}
          HETZNER_USER: ${{ secrets.HETZNER_USER }}
          HETZNER_HOST: ${{ secrets.HETZNER_HOST }}
          REMOTE_DIR: ${{ secrets.REMOTE_DIR }}
        run: |
          ssh -p "${HETZNER_SSH_PORT:-22}" "${HETZNER_USER}@${HETZNER_HOST}" "
            set -euo pipefail
            cd \"${REMOTE_DIR:-/opt/lbpr}\"
            export DOCKERHUB_USERNAME='${{ secrets.DOCKERHUB_USERNAME }}'
            export API_TAG='${{ steps.tags.outputs.api_tag }}'
            export SPA_TAG='${{ steps.tags.outputs.spa_tag }}'

            echo '${{ secrets.DOCKERHUB_TOKEN }}' | docker login --username '${{ secrets.DOCKERHUB_USERNAME }}' --password-stdin || true

            echo 'DOPPLER_PROJECT=${{ secrets.DOPPLER_PROJECT }}' > .deploy.env
            echo 'DOPPLER_CONFIG=${{ secrets.DOPPLER_CONFIG }}' >> .deploy.env
            echo 'DOPPLER_ENABLED=1' >> .deploy.env

            bash ops/deploy/deploy.sh
          "

name: Infra (Hetzner) — plan & apply + deploy

on:
  workflow_dispatch:
    inputs:
      apply:
        description: "Apply after plan?"
        type: boolean
        default: true

jobs:
  terraform:
    runs-on: ubuntu-latest
    defaults:
      run:
        working-directory: infra/terraform/hetzner
    steps:
      - uses: actions/checkout@v4

      - uses: hashicorp/setup-terraform@v3
        with:
          terraform_version: 1.11.4

      - name: Write terraform.tfvars from GitHub Secrets (example)
        run: |
          cat > terraform.tfvars <<'EOF'
          hcloud_token   = "${{ secrets.HCLOUD_TOKEN }}"
          ssh_public_key = "${{ secrets.DEPLOY_SSH_PUBKEY }}"
          # Optional: lock down SSH to CI runner public IP(s)
          # ssh_allowed_cidrs = ["${{ secrets.SSH_ALLOWED_CIDR }}"]
          EOF

      - name: Terraform Init
        run: terraform init

      - name: Terraform Validate
        run: terraform validate

      - name: Terraform Plan
        id: plan
        run: terraform plan -no-color

      - name: Terraform Apply
        if: ${{ inputs.apply }}
        run: terraform apply -auto-approve

      - name: Export server IP output
        if: ${{ inputs.apply }}
        id: out
        run: echo "IP=$(terraform output -raw server_ipv4)" >> $GITHUB_OUTPUT

  deploy:
    needs: terraform
    if: ${{ inputs.apply }}
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4

      - name: Configure SSH
        run: |
          mkdir -p ~/.ssh
          echo "${{ secrets.DEPLOY_SSH_KEY }}" > ~/.ssh/id_ed25519
          chmod 600 ~/.ssh/id_ed25519
          # Add server host key (best: use known_hosts from a trusted source)
          ssh-keyscan -H "${{ needs.terraform.outputs.IP }}" >> ~/.ssh/known_hosts

      - name: Rsync app bundle
        run: |
          rsync -az --delete \
            -e 'ssh -i ~/.ssh/id_ed25519' \
            ./ \
            "deploy@${{ needs.terraform.outputs.IP }}:/opt/lbpr/"

      - name: Remote deploy via deploy.sh
        env:
          DOCKERHUB_USERNAME: ${{ secrets.DOCKERHUB_USERNAME }}
        run: |
          ssh -i ~/.ssh/id_ed25519 "deploy@${{ needs.terraform.outputs.IP }}" "
            set -euo pipefail
            cd /opt/lbpr
            export DOCKERHUB_USERNAME='${DOCKERHUB_USERNAME}'
            bash ops/deploy/deploy.sh
          "
